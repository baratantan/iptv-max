async function fetchWithProxies(url, logElement, timeoutMs = 5000) { if (!url) { throw new Error('الرابط (URL) مطلوب'); } const encodedUrl = encodeURIComponent(url); // تم ترتيب الوكلاء من الأكثر موثوقية إلى الأقل const proxies = [ { name: 'AllOrigins', url: `https://api.allorigins.win/raw?url=${encodedUrl}` }, { name: 'CORS Proxy', url: `https://corsproxy.io/?${encodedUrl}` }, { name: 'ThingProxy', url: `https://thingproxy.freeboard.io/fetch/${encodedUrl}` }, { name: 'CORS.sh', url: `https://proxy.cors.sh/${encodedUrl}`, headers: { 'x-requested-with': 'XMLHttpRequest' } } ]; for (let i = 0; i < proxies.length; i++) { const proxy = proxies[i]; const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), timeoutMs); try { if (logElement) { logElement.insertAdjacentHTML('beforeend', `
🔄 محاولة ${i + 1}: ${proxy.name}...
`); logElement.scrollTop = logElement.scrollHeight; } const options = { method: 'GET', signal: controller.signal // ربط إشارة الإلغاء بطلب fetch }; if (proxy.headers) { options.headers = proxy.headers; } const response = await fetch(proxy.url, options); clearTimeout(timeoutId); // إلغاء المؤقت لأن الطلب نجح if (!response.ok) { throw new Error(`HTTP ${response.status}`); } const content = await response.text(); // التحقق من أن المحتوى ليس فارغاً أو صفحة خطأ صغيرة من الوكيل if (!content || content.trim().length < 20) { throw new Error('المحتوى المسترجع فارغ أو غير صالح'); } if (logElement) { logElement.insertAdjacentHTML('beforeend', `
✅ نجح: ${proxy.name} (${(content.length/1024).toFixed(1)} KB)
`); logElement.scrollTop = logElement.scrollHeight; } return content; } catch (error) { clearTimeout(timeoutId); // تنظيف المؤقت في حالة الخطأ const errorMsg = error.name === 'AbortError' ? 'انتهت مهلة الانتظار (Timeout)' : error.message; if (logElement) { logElement.insertAdjacentHTML('beforeend', `
❌ فشل: ${proxy.name} - ${errorMsg}
`); logElement.scrollTop = logElement.scrollHeight; } } } throw new Error('فشلت جميع محاولات التحميل عبر الوكلاء (Proxies)'); }
